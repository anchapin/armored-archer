package quality

import (
	"bytes"
	"os"
	"path/filepath"
	"strings"
)

// Assertion patterns to search for in test files
var assertionPatterns = []string{
	"assert.Equal",
	"assert.True",
	"assert.False",
	"assert.NotNil",
	"assert.Nil",
	"assert.Error",
	"assert.NoError",
	"assert.Contains",
	"assert.Len",
	"assert.Panics",
	"require.Equal",
	"require.True",
	"require.False",
	"require.NotNil",
	"require.Nil",
	"require.Error",
	"require.NoError",
	"require.Contains",
	"require.Len",
}

// HasAssertions checks if a test file contains at least one assertion
func HasAssertions(testFile string) (bool, error) {
	content, err := os.ReadFile(testFile)
	if err != nil {
		return false, err
	}

	for _, pattern := range assertionPatterns {
		if bytes.Contains(content, []byte(pattern)) {
			return true, nil
		}
	}

	return false, nil
}

// CountAssertions counts the number of assertions in a test file
func CountAssertions(testFile string) (int, error) {
	content, err := os.ReadFile(testFile)
	if err != nil {
		return 0, err
	}

	count := 0
	for _, pattern := range assertionPatterns {
		count += bytes.Count(content, []byte(pattern))
	}

	return count, nil
}

// FindTestFiles recursively finds all Go test files in a directory
func FindTestFiles(rootDir string) ([]string, error) {
	var testFiles []string

	err := filepath.Walk(rootDir, func(path string, info os.FileInfo, err error) error {
		if err != nil {
			return err
		}
		if info.IsDir() {
			// Skip vendor and hidden directories
			base := filepath.Base(path)
			if base == "vendor" || strings.HasPrefix(base, ".") {
				return filepath.SkipDir
			}
			return nil
		}
		if strings.HasSuffix(path, "_test.go") {
			testFiles = append(testFiles, path)
		}
		return nil
	})

	return testFiles, err
}

// CheckAllTests checks all test files in a directory for assertions
// Returns a list of test files without assertions
func CheckAllTests(rootDir string) ([]string, error) {
	testFiles, err := FindTestFiles(rootDir)
	if err != nil {
		return nil, err
	}

	var filesWithoutAssertions []string
	for _, file := range testFiles {
		hasAssertions, err := HasAssertions(file)
		if err != nil {
			return nil, err
		}
		if !hasAssertions {
			filesWithoutAssertions = append(filesWithoutAssertions, file)
		}
	}

	return filesWithoutAssertions, nil
}

package quality

import (
	"os"
	"path/filepath"
	"testing"

	"github.com/stretchr/testify/assert"
	"github.com/stretchr/testify/require"
)

func TestHasAssertions(t *testing.T) {
	tests := []struct {
		name     string
		content  string
		expected bool
	}{
		{
			name:     "file with assertion",
			content:  "func TestSomething(t *testing.T) {\n\tassert.Equal(t, 1, 1)\n}\n",
			expected: true,
		},
		{
			name:     "file with require assertion",
			content:  "func TestSomething(t *testing.T) {\n\trequire.NoError(t, err)\n}\n",
			expected: true,
		},
		{
			name:     "file without assertions",
			content:  "func TestSomething(t *testing.T) {\n\t_ = 1\n}\n",
			expected: false,
		},
		{
			name:     "empty file",
			content:  "",
			expected: false,
		},
	}

	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			// Create temporary file
			tmpDir := t.TempDir()
			tmpFile := filepath.Join(tmpDir, "test.go")
			err := os.WriteFile(tmpFile, []byte(tt.content), 0644)
			require.NoError(t, err)

			// Check assertions
			hasAssertions, err := HasAssertions(tmpFile)
			require.NoError(t, err)
			assert.Equal(t, tt.expected, hasAssertions)
		})
	}
}

func TestCountAssertions(t *testing.T) {
	content := `func TestSomething(t *testing.T) {
		assert.Equal(t, 1, 1)
		assert.True(t, true)
		assert.Equal(t, 2, 2)
	}`

	tmpDir := t.TempDir()
	tmpFile := filepath.Join(tmpDir, "test.go")
	err := os.WriteFile(tmpFile, []byte(content), 0644)
	require.NoError(t, err)

	count, err := CountAssertions(tmpFile)
	require.NoError(t, err)
	assert.Equal(t, 3, count)
}

func TestFindTestFiles(t *testing.T) {
	tmpDir := t.TempDir()

	// Create test files
	os.WriteFile(filepath.Join(tmpDir, "test_one_test.go"), []byte("test"), 0644)
	os.WriteFile(filepath.Join(tmpDir, "test_two_test.go"), []byte("test"), 0644)
	os.WriteFile(filepath.Join(tmpDir, "not_test.go"), []byte("test"), 0644)

	// Create subdirectory with test
	subDir := filepath.Join(tmpDir, "sub")
	os.Mkdir(subDir, 0755)
	os.WriteFile(filepath.Join(subDir, "sub_test.go"), []byte("test"), 0644)

	files, err := FindTestFiles(tmpDir)
	require.NoError(t, err)
	assert.Len(t, files, 3) // Should find 3 _test.go files
}

func TestCheckAllTests(t *testing.T) {
	tmpDir := t.TempDir()

	// Create test file with assertion
	os.WriteFile(filepath.Join(tmpDir, "good_test.go"), []byte("func Test(t *testing.T) {\n\tassert.Equal(t, 1, 1)\n}\n"), 0644)

	// Create test file without assertion
	os.WriteFile(filepath.Join(tmpDir, "bad_test.go"), []byte("func Test(t *testing.T) {\n\t_ = 1\n}\n"), 0644)

	// Create non-test file
	os.WriteFile(filepath.Join(tmpDir, "not_test.go"), []byte("func Test(t *testing.T) {\n\t_ = 1\n}\n"), 0644)

	files, err := CheckAllTests(tmpDir)
	require.NoError(t, err)
	assert.Len(t, files, 1)
	assert.Contains(t, files[0], "bad_test.go")
}
