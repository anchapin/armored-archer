2. Create metrics registry at package level (after globalCache declaration):
   ```go
   var (
       // Latency histogram for RPC calls
       rpcLatency = prometheus.NewHistogramVec(
           prometheus.HistogramOpts{
               Name:    "nakama_rpc_latency_seconds",
               Help:    "RPC call latency in seconds",
               Buckets: []float64{0.01, 0.025, 0.05, 0.1, 0.25, 0.5, 1, 2.5, 5, 10},
           },
           []string{"rpc_method", "status"},
       )

       // Error rate counter
       rpcErrors = prometheus.NewCounterVec(
           prometheus.CounterOpts{
               Name: "nakama_rpc_errors_total",
               Help: "Total number of RPC errors",
           },
           []string{"rpc_method", "error_type"},
       )

       // Active connections gauge
       activeConnections = prometheus.NewGauge(
           prometheus.GaugeOpts{
               Name: "nakama_active_connections",
               Help: "Number of active database connections",
           },
       )
   )
   ```

3. Register metrics in InitModule:
   ```go
   func InitModule(...) error {
       // ... existing code ...

       // Register Prometheus metrics
       prometheus.MustRegister(rpcLatency)
       prometheus.MustRegister(rpcErrors)
       prometheus.MustRegister(activeConnections)

       // Start metrics server in goroutine
       go func() {
           metricsPort := ":9090" // Or read from config
           http.Handle("/metrics", promhttp.Handler())
           logger.Info("Prometheus metrics server listening on %s", metricsPort)
           if err := http.ListenAndServe(metricsPort, nil); err != nil {
               logger.Error("Metrics server error: %v", err)
           }
       }()

       // ... rest of InitModule ...
   }
   ```

4. Export metrics accessor for RPC handlers:
   ```go
   // RecordRPCLatency records RPC call latency
   func RecordRPCLatency(method string, status string, duration time.Duration) {
       rpcLatency.WithLabelValues(method, status).Observe(duration.Seconds())
   }

   // RecordRPCError records an RPC error
   func RecordRPCError(method string, errorType string) {
       rpcErrors.WithLabelValues(method, errorType).Inc()
   }
   ```

Add "time" import if not already present.

   ```go
   func NewCacheManager(logger runtime.Logger) *CacheManager {
       return &CacheManager{
           logger: logger,
           caches: make(map[string]*LRUCache),

           cacheHits: prometheus.NewCounterVec(
               prometheus.CounterOpts{
                   Name: "cache_hits_total",
                   Help: "Total number of cache hits",

           ),

           cacheMisses: prometheus.NewCounterVec(
               prometheus.CounterOpts{
                   Name: "cache_misses_total",
                   Help: "Total number of cache misses",
               },
               []string{"cache_name"},
           ),

           cacheHitRate: prometheus.NewGaugeVec(
               prometheus.GaugeOpts{
                   Name: "cache_hit_rate",
                   Help: "Cache hit rate (0-1)",
               },
               []string{"cache_name"},
           ),
       }
   }
   ```

4. Register metrics in main.go InitModule after cache creation:
   ```go
   prometheus.MustRegister(globalCache.CacheHits())
   prometheus.MustRegister(globalCache.CacheMisses())
   prometheus.MustRegister(globalCache.CacheHitRate())
   ```

5. Add metric collection methods to CacheManager:
   ```go
   // CacheHits returns the cache hits counter
   func (m *CacheManager) CacheHits() *prometheus.CounterVec {
       return m.cacheHits
   }

   // CacheMisses returns the cache misses counter
   func (m *CacheManager) CacheMisses() *prometheus.CounterVec {
       return m.cacheMisses
   }

   // CacheHitRate returns the cache hit rate gauge
   func (m *CacheManager) CacheHitRate() *prometheus.GaugeVec {
       return m.cacheHitRate
   }

   // UpdateMetrics updates hit rate gauges for all caches
   func (m *CacheManager) UpdateMetrics() {
       m.mu.RLock()
       defer m.mu.RUnlock()

       for name, cache := range m.caches {
           stats := cache.Stats()
           total := stats.Hits + stats.Misses
           if total > 0 {
               hitRate := float64(stats.Hits) / float64(total)
               m.cacheHitRate.WithLabelValues(name).Set(hitRate)
           }
       }
   }
   ```
