        ctrl := gomock.NewController(t)
        mockLogger := mocks.NewMockNakamaLogger(ctrl)
        manager := utils.NewCacheManager(mockLogger)
        cache := manager.CreateCache("test_cache", 100, time.Minute)

        // Add some items
        for i := 0; i < 50; i++ {
            cache.Set(fmt.Sprintf("key%d", i), fmt.Sprintf("value%d", i), time.Minute)
        }

        // Concurrent reads
        var wg sync.WaitGroup
        readCount := 0
        for i := 0; i < 100; i++ {
            wg.Add(1)
            go func() {
                defer wg.Done()
                val, ok := cache.Get("key25")
                if ok {
                    if val == "value25" {
                        readCount++
                    }
                }
            }()
        }
        wg.Wait()
