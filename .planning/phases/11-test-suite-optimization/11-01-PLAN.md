    Add pyramid validation step to .github/workflows/test.yml after backend-tests job but before require-tests gate:

    Insert new job "test-pyramid-validation" before "require-tests":
    ```yaml
  test-pyramid-validation:
    name: Test Pyramid Validation
    runs-on: ubuntu-latest
    needs: [backend-tests, godot-tests]
    steps:
      - uses: actions/checkout@v4

      - name: Run test pyramid validation
        id: pyramid
        run: |
          ./scripts/check-test-pyramid.sh --ci
          # Capture CI output for GITHUB_OUTPUT
          echo "total=$(grep TEST_PYRAMID_TOTAL <<< '$(./scripts/check-test-pyramid.sh --ci)' | cut -d= -f2)" >> $GITHUB_OUTPUT
          echo "unit_pct=$(grep TEST_PYRAMID_UNIT_PCT <<< '$(./scripts/check-test-pyramid.sh --ci)' | cut -d= -f2)" >> $GITHUB_OUTPUT
          echo "integration_pct=$(grep TEST_PYRAMID_INTEGRATION_PCT <<< '$(./scripts/check-test-pyramid.sh --ci)' | cut -d= -f2)" >> $GITHUB_OUTPUT
          echo "e2e_pct=$(grep TEST_PYRAMID_E2E_PCT <<< '$(./scripts/check-test-pyramid.sh --ci)' | cut -d= -f2)" >> $GITHUB_OUTPUT

      - name: Upload pyramid results
        uses: actions/upload-artifact@v4
        with:
          name: test-pyramid-results
          path: data/test-pyramid-results.json
          retention-days: 30

      - name: Comment on PR
        if: github.event_name == 'pull_request'
        uses: actions/github-script@v7
        with:
          script: |
            const total = '${{ steps.pyramid.outputs.total }}';
            const unit = '${{ steps.pyramid.outputs.unit_pct }}';
            const integration = '${{ steps.pyramid.outputs.integration_pct }}';
            const e2e = '${{ steps.pyramid.outputs.e2e_pct }}';
            let body = '## 📊 Test Pyramid Validation\n\n';
            body += '| Type | Count | % | Target | Status |\n';
            body += '|------|-------|---|--------|--------|\n';
            body += `| Unit | ${Math.round(total * unit / 100)} | ${unit}% | 70% ±10% | ${unit >= 60 && unit <= 80 ? '✅' : '❌'} |\n`;
            body += `| Integration | ${Math.round(total * integration / 100)} | ${integration}% | 20% ±10% | ${integration >= 10 && integration <= 30 ? '✅' : '❌'} |\n`;
            body += `| E2E | ${Math.round(total * e2e / 100)} | ${e2e}% | 10% ±10% | ${e2e <= 20 ? '✅' : '❌'} |\n`;
            body += '\n_This comment was automatically generated by test pyramid validation._';
            github.rest.issues.createComment({
              issue_number: context.issue.number,
              owner: context.repo.owner,
              repo: context.repo.repo,
              body: body
            });
    ```

    Update "require-tests" job to include pyramid validation:
    ```yaml
  require-tests:
    name: All Tests Must Pass
    runs-on: ubuntu-latest
    needs: [godot-tests, godot-coverage-gate, backend-tests, test-pyramid-validation]
    # ... existing steps, add pyramid check
    - name: Check if all validations passed
      run: |
        # ... existing checks ...
        PYRAMID_RESULT=${{ needs.test-pyramid-validation.result }}
        echo "Pyramid validation: $PYRAMID_RESULT"
        if [ "$PYRAMID_RESULT" != "success" ] && [ "$PYRAMID_RESULT" != "skipped" ]; then
          echo "❌ Test pyramid validation failed"
          exit 1
        fi
    ```
  </action>
