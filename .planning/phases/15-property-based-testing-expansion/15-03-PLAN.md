```go
// loadout represents a player's equipment (5 slots)
type loadout struct {
    helm   string // gear_id or empty string
    armor  string
    bow    string
    arrow  string
    amulet string
}

// canEquip checks if gear can be equipped to loadout (slot uniqueness)
func canEquip(l *loadout, gearID, gearType string) bool {
    switch gearType {
    case "helm":
        return l.helm == "" || l.helm == gearID
    case "armor":
        return l.armor == "" || l.armor == gearID
    case "bow":
        return l.bow == "" || l.bow == gearID
    case "arrow":
        return l.arrow == "" || l.arrow == gearID
    case "amulet":
        return l.amulet == "" || l.amulet == gearID
    default:
        return false // Invalid gear type
    }
}

// equipGear simulates equipping gear to loadout
func equipGear(l *loadout, gearID, gearType string) error {
    if !canEquip(l, gearID, gearType) {
        return fmt.Errorf("cannot equip gear type %s", gearType)
    }
    switch gearType {
    case "helm":
        l.helm = gearID
    case "armor":
        l.armor = gearID
    case "bow":
        l.bow = gearID
    case "arrow":
        l.arrow = gearID
    case "amulet":
        l.amulet = gearID
    }
    return nil
}

// slotCount returns number of non-empty slots in loadout
func slotCount(l *loadout) int {
    count := 0
    if l.helm != "" { count++ }
    if l.armor != "" { count++ }
    if l.bow != "" { count++ }
    if l.arrow != "" { count++ }
    if l.amulet != "" { count++ }
    return count
}
```
