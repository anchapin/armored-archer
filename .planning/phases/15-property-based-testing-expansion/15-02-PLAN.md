```go
// shouldMatch returns true if two players should match based on skill tolerance
func shouldMatch(skill1, skill2 int, tolerance float64) bool {
    if skill1 == 0 || skill2 == 0 {
        return false // Invalid skill values
    }
    ratio1 := float64(skill1) / float64(skill2)
    ratio2 := float64(skill2) / float64(skill1)
    return ratio1 >= 1-tolerance && ratio1 <= 1+tolerance &&
           ratio2 >= 1-tolerance && ratio2 <= 1+tolerance
}
```
