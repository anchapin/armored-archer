import xml.etree.ElementTree as ET
tree = ET.parse('test/results/gut-results.xml')
root = tree.getroot()
tests = root.findall('.//testcase')
autoload_tests = [t for t in tests if 'autoload' in t.get('classname', '')]

errors = len([t for t in autoload_tests if t.find('error') is not None])
pass_rate = ((len(autoload_tests) - failures - errors) / len(autoload_tests) * 100) if autoload_tests else 0
print(f'Autoload pass rate: {pass_rate:.1f}%')
print(f'Autoload failures: {failures}, errors: {errors}')
exit(0 if pass_rate >= 95.0 else 1)
"

    Add a new entry to data/coverage-history.json with the current Godot pass rate:

    ```bash
    # Get current pass rate
    PASS_RATE=$(python3 scripts/calculate_godot_coverage.py test/results/gut-results.xml | jq -r '.pass_rate')
    TOTAL_TESTS=$(python3 scripts/calculate_godot_coverage.py test/results/gut-results.xml | jq -r '.total_tests')

    # Get current timestamp in ISO format
    TIMESTAMP=$(date -u +"%Y-%m-%dT%H:%M:%SZ")

    # Add new entry to coverage-history.json
    jq --arg ts "$TIMESTAMP" --argjson pr "$PASS_RATE" --argjson tt "$TOTAL_TESTS" '. += [{"timestamp": $ts, "godot_pass_rate": $pr, "godot_total_tests": $tt}]' data/coverage-history.json > data/coverage-history.json.tmp
    mv data/coverage-history.json.tmp data/coverage-history.json
    ```

    The new entry format:
    ```json
    {
        "timestamp": "2026-03-21T12:00:00Z",
        "godot_pass_rate": 95.5,
        "godot_total_tests": 150
    }
    ```

    If data/coverage-history.json doesn't exist, create it with an empty array first.
  </action>
