## Technical Notes

### Applying Design Tokens

```gdscript
# Example: Applying design tokens to a button
func _apply_design_system() -> void:
    var colors = ThemeManager.get_theme_colors()
    
    # Background
    modulate = colors["surface"]
    
    # Use design tokens for specific elements
    primary_button.modulate = DesignTokens.COLOR_PRIMARY
    gold_label.modulate = DesignTokens.COLOR_GOLD
    gem_label.modulate = DesignTokens.COLOR_GEMS
    
    # Health bar
    health_bar.modulate = DesignTokens.COLOR_HEALTH
    
    # Rarity colors
    common_item.modulate = DesignTokens.COLOR_RARITY_COMMON
    rare_item.modulate = DesignTokens.COLOR_RARITY_RARE
    epic_item.modulate = DesignTokens.COLOR_RARITY_EPIC
    legendary_item.modulate = DesignTokens.COLOR_RARITY_LEGENDARY
```
