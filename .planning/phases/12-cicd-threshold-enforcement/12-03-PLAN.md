```bash
#!/bin/bash
set -e

# Coverage Gates Enforcement Script
# Enforces coverage thresholds with incremental stages
# Usage: coverage_gates.sh [critical]
#   critical - Only enforce critical path threshold (skip overall check)
#
# Environment variables:
#   COVERAGE_GATE_STAGE - Gate stage to enforce (1=30%, 2=45%, 3=60%, default=1)

cd "$(dirname "$0")/../.."

# Colors for output
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
NC='\033[0m' # No Color

# Configuration files
THRESHOLDS_FILE="data/coverage-thresholds.json"
BASELINES_FILE="data/package-baselines.json"
COVERAGE_FILE="coverage/coverage.out"

# Load thresholds from configuration
if [ ! -f "$THRESHOLDS_FILE" ]; then
    echo -e "${RED}ERROR: $THRESHOLDS_FILE not found${NC}"
    exit 1
fi

OVERALL_THRESHOLD=$(jq -r '.overall' "$THRESHOLDS_FILE")
CRITICAL_THRESHOLD=$(jq -r '.critical' "$THRESHOLDS_FILE")
GODOT_THRESHOLD=$(jq -r '.godot_pass_rate_threshold' "$THRESHOLDS_FILE")
REGRESSION_WARNING=$(jq -r '.regression_warning' "$THRESHOLDS_FILE")

# Get current gate stage from environment or default to 1
THRESHOLD_STAGE="${COVERAGE_GATE_STAGE:-1}"

# Get threshold for current stage
STAGE_THRESHOLD=$(jq -r ".stages[\"$THRESHOLD_STAGE\"] // .overall" "$THRESHOLDS_FILE")

# Get critical packages list
CRITICAL_PACKAGES=$(jq -r '.critical_packages | join(" ")' "$THRESHOLDS_FILE")

# Parse arguments
MODE="all"
if [ "$1" = "critical" ]; then
    MODE="critical"
fi

echo "=== Coverage Gates Enforcement ==="
echo -e "${BLUE}Mode:${NC} ${MODE}"
echo -e "${BLUE}Gate Stage:${NC} ${THRESHOLD_STAGE} (target: ${STAGE_THRESHOLD}%)"
echo -e "${BLUE}Overall threshold:${NC} ${OVERALL_THRESHOLD}% (final target)"
echo -e "${BLUE}Critical path threshold:${NC} ${CRITICAL_THRESHOLD}%"
echo -e "${BLUE}Critical packages:${NC} ${CRITICAL_PACKAGES}"
echo -e "${BLUE}Godot pass rate threshold:${NC} ${GODOT_THRESHOLD}%"
echo ""

# Check if coverage file exists
if [ ! -f "$COVERAGE_FILE" ]; then
    echo -e "${RED}ERROR: $COVERAGE_FILE not found${NC}"
    exit 1
fi

# Extract overall coverage
OVERALL=$(go tool cover -func="$COVERAGE_FILE" | grep "^total:" | awk '{print $3}' | sed 's/%//')
OVERALL_NUM=$(echo "$OVERALL" | awk '{printf "%.1f", $1}')

echo -e "${BLUE}Overall coverage:${NC} ${OVERALL_NUM}%"

# Check overall threshold based on gate stage
if [ "$MODE" = "all" ]; then
    if (( $(echo "$OVERALL_NUM < $STAGE_THRESHOLD" | bc -l 2>/dev/null || echo "0") )); then
        echo -e "${RED}FAIL: Overall coverage ${OVERALL_NUM}% is below stage ${THRESHOLD_STAGE} threshold ${STAGE_THRESHOLD}%${NC}"
        echo -e "${YELLOW}To advance to next stage, set COVERAGE_GATE_STAGE=$(($THRESHOLD_STAGE + 1))${NC}"
        exit 1
    fi

    echo -e "${GREEN}PASS: Overall coverage ${OVERALL_NUM}% meets stage ${THRESHOLD_STAGE} threshold${NC}"

    # Show progress toward next stage
    if [ "$THRESHOLD_STAGE" -lt 3 ]; then
        NEXT_STAGE_THRESHOLD=$(jq -r ".stages[\"$(($THRESHOLD_STAGE + 1))\"]" "$THRESHOLDS_FILE")
        PROGRESS=$(echo "scale=1; ($OVERALL_NUM - $STAGE_THRESHOLD) * 100 / ($NEXT_STAGE_THRESHOLD - $STAGE_THRESHOLD)" | bc 2>/dev/null || echo "0")
        echo -e "${BLUE}Progress to stage $(($THRESHOLD_STAGE + 1)): ${PROGRESS}%${NC}"
    fi
fi

# Check critical packages
echo ""
echo -e "${BLUE}Critical package coverage:${NC}"

CRITICAL_MET=1
for pkg in $CRITICAL_PACKAGES; do
    # Extract coverage for specific package
    pkg_lines=$(go tool cover -func="$COVERAGE_FILE" | grep "internal/$pkg/" | grep -v "^total:")

    if [ -z "$pkg_lines" ]; then
        echo -e "  ${YELLOW}internal/$pkg:${NC} No coverage data (not tested yet)"
        continue
    fi

    # Calculate average coverage for this package
    pkg_cov=$(echo "$pkg_lines" | awk '{
        coverage = $3
        gsub(/%/, "", coverage)
        sum += coverage
        count++
    } END {
        if (count > 0) printf "%.1f", sum/count
        else print "0.0"
    }')

    echo -n "  internal/$pkg: ${pkg_cov}%"

    # Check critical threshold
    if (( $(echo "$pkg_cov < $CRITICAL_THRESHOLD" | bc -l 2>/dev/null || echo "1") )); then
        echo -e " ${RED}(FAIL - threshold: ${CRITICAL_THRESHOLD}%)${NC}"
        CRITICAL_MET=0
    else
        echo -e " ${GREEN}(PASS)${NC}"
    fi
done

echo ""

# Check package-level regressions against baselines
echo -e "${BLUE}Package regression check:${NC}"

if [ -f "$BASELINES_FILE" ]; then
    REGRESSIONS_FOUND=0

    # Extract all packages from coverage
    go tool cover -func="$COVERAGE_FILE" | grep -v "^total:" | awk '{
        split($1, parts, ":")
        pkg = parts[1]
        gsub(/\/[^\/]+$/, "", pkg)
        coverage = $3
        gsub(/%/, "", coverage)
        pkg_count[pkg]++
        pkg_total[pkg] += coverage
    } END {
        for (p in pkg_count) {
            avg = pkg_total[p] / pkg_count[p]
            printf "%s|%.1f\n", p, avg
        }
    }' | while IFS='|' read -r pkg coverage; do
        BASELINE=$(jq -r ".packages.\"$pkg\" // \"0.0\"" "$BASELINES_FILE")

        # Skip packages with no baseline data
        if [ "$BASELINE" = "null" ] || [ -z "$BASELINE" ]; then
            continue
        fi

        # Calculate difference
        DIFF=$(echo "$coverage - $BASELINE" | bc -l 2>/dev/null || echo "0")

        # Check for regression
        if (( $(echo "$DIFF < -$REGRESSION_WARNING" | bc -l 2>/dev/null || echo "0") )); then
            PKG_NAME="${pkg##*/}"
            echo -e "  ${YELLOW}${PKG_NAME}${NC}: ${coverage}% (baseline: ${BASELINE}%, diff: ${DIFF}%) ${YELLOW}[REGRESSION WARNING]${NC}"
            REGRESSIONS_FOUND=1
        fi
    done
else
    echo "  ${YELLOW}No baseline file found, skipping regression check${NC}"
fi

# Exit with failure if any critical package below threshold
if [ $CRITICAL_MET -eq 0 ]; then
    echo ""
    echo -e "${RED}FAIL: One or more critical packages below threshold${NC}"
    exit 1
fi

echo ""
echo -e "${GREEN}All coverage gates passed${NC}"
exit 0
```

3. Show progress bar for next stage advancement
4. Check package-level regressions against baselines
5. Support Godot pass rate threshold from configuration
6. Keep critical-only mode for incremental progress

```makefile
# Coverage Gates
.PHONY: coverage-gates
coverage-gates: ## Enforce coverage thresholds (uses COVERAGE_GATE_STAGE env var, default 1)
	@echo "Enforcing coverage gates (stage $${COVERAGE_GATE_STAGE:-1})..."
	@cd backend/tests/quality && ./coverage_gates.sh

.PHONY: coverage-gate-critical
coverage-gate-critical: ## Enforce only critical path threshold (skip overall check)
	@cd backend/tests/quality && ./coverage_gates.sh critical

.PHONY: coverage-threshold-check
coverage-threshold-check: ## Show current coverage against all threshold stages
	@echo "=== Coverage Threshold Status ==="
	@echo ""
	@cd backend && go test -coverprofile=coverage/coverage.out -covermode=atomic -coverpkg=./internal/... ./... > /dev/null 2>&1
	@OVERALL=$$(go tool cover -func=coverage/coverage.out | grep total | awk '{print $$3}' | sed 's/%//'); \
	 echo "Current Overall Coverage: $$OVERALL%"; \
	 echo ""; \
	 echo "Stage Thresholds:"; \
	 echo "  Stage 1 (30%): $$(echo "$$OVERALL < 30" | bc -l 2>/dev/null || echo 0 | awk '{if($$1==1) print "FAIL"; else print "PASS"}')"; \
	 echo "  Stage 2 (45%): $$(echo "$$OVERALL < 45" | bc -l 2>/dev/null || echo 0 | awk '{if($$1==1) print "FAIL"; else print "PASS"}')"; \
	 echo "  Stage 3 (60%): $$(echo "$$OVERALL < 60" | bc -l 2>/dev/null || echo 0 | awk '{if($$1==1) print "FAIL"; else print "PASS"}')"; \
	 echo ""; \
	 echo "Critical Path Threshold (80%):"; \
	 CRITICAL=$$(go tool cover -func=coverage/coverage.out | grep -E "(combat|matchmaking|rpg)" | awk '{gsub(/%/, "", $$3); sum+=$$3; n++} END {printf "%.1f", n>0 ? sum/n : "0"}'); \
	 echo "  Current: $$CRITICAL%"; \
	 echo "  Status: $$(echo "$$CRITICAL < 80" | bc -l 2>/dev/null || echo 1 | awk '{if($$1==1) print "FAIL"; else print "PASS"}')"

.PHONY: coverage-baseline-update
coverage-baseline-update: ## Update package baselines from current coverage
	@cd backend && go test -coverprofile=coverage/coverage.out -covermode=atomic -coverpkg=./internal/... ./... > /dev/null 2>&1
	@scripts/check-coverage-regression.sh --update-baselines
	@echo "Baselines updated"

.PHONY: coverage-gap-analysis
coverage-gap-analysis: ## Show coverage gaps (0% coverage functions)
	@cd backend && go test -coverprofile=coverage/coverage.out -covermode=atomic -coverpkg=./internal/... ./... > /dev/null 2>&1
	@scripts/gap-analysis.sh

.PHONY: coverage-history
coverage-history: ## Track coverage in history and show trend
	@cd backend && go test -coverprofile=coverage.out -covermode=atomic -coverpkg=./internal/... ./... > /dev/null 2>&1
	@scripts/track-coverage-history.sh
```
