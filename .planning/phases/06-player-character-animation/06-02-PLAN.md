2. Add animation state enum:
   enum AnimState { IDLE, WALK, BOW_DRAW, ATTACK, HIT, DEATH }
   var current_anim_state := AnimState.IDLE

3. Add state priority constant:
   const STATE_PRIORITY := {
       AnimState.DEATH: 5,
       AnimState.HIT: 4,
       AnimState.ATTACK: 3,
       AnimState.BOW_DRAW: 2,
       AnimState.WALK: 1,
       AnimState.IDLE: 0,
   }

4. Add direction helper function:
   func _get_direction_suffix(dir: Vector2) -> String:
       if abs(dir.x) > abs(dir.y):
           return "right" if dir.x > 0 else "left"
       else:
           return "down" if dir.y > 0 else "up"

5. Add animation update function that respects priority:
   func _update_animation() -> void:
       var direction_suffix := _get_direction_suffix(aim_direction)
       var anim_name := ""
       
       match current_anim_state:
           AnimState.IDLE:
               anim_name = "idle_" + direction_suffix
           AnimState.WALK:
               anim_name = "walk_" + direction_suffix
           AnimState.BOW_DRAW:
               anim_name = "bow_draw_" + direction_suffix
           AnimState.ATTACK:
               anim_name = "attack_" + direction_suffix
           AnimState.HIT:
               anim_name = "hit_" + direction_suffix
           AnimState.DEATH:
               anim_name = "death_" + direction_suffix
       
       if animated_sprite.sprite_frames.has_animation(anim_name):
           animated_sprite.play(anim_name)

6. Add state change logic in _physics_process:
   - If is_aiming and STATE_PRIORITY[BOW_DRAW] >= current priority: change to BOW_DRAW
   - If is_moving and STATE_PRIORITY[WALK] >= current priority: change to WALK  
   - Otherwise: change to IDLE

7. Connect animation_finished signal to handle one-shot animations

               current_anim_state = AnimState.IDLE
               _update_animation()
           AnimState.DEATH:
               animated_sprite.stop()  # Stay on last frame
