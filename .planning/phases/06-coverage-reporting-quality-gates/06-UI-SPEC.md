## Spacing Scale

Declared values (8-point scale, all multiples of 4):

| Token | Value | Usage |
|-------|-------|-------|
| xxs | 2px | Micro adjustments, hairline gaps |
| xs | 4px | Icon gaps, inline padding, tight spacing |
| sm | 8px | Compact element spacing, icon-button padding |
| md | 12px | Default element spacing, form inputs |
| lg | 16px | Section padding, card spacing |
| xl | 24px | Layout gaps, major section spacing |
| 2xl | 32px | Large section breaks, page margins |
| 3xl | 48px | Page-level spacing, major content blocks |

**Animations disabled in test mode:** Set `Engine.physics_ticks_per_second = 0` and `OS.low_processor_usage_mode = true` to freeze animations for stable screenshots.

**Screenshot capture pattern:**
```gdscript
# test/suites/visual/test_visual_regression.gd

func test_component_visual_regression():
    # Disable animations for stable screenshots
    Engine.physics_ticks_per_second = 0

    # Set viewport size
    get_viewport().size = Vector2(375, 667)  # Mobile

    # Capture screenshot
    var viewport = get_viewport()
    var image = viewport.get_texture().get_data()
    image.flip_y()  # Godot textures are flipped

    # Save current screenshot
    var current_path = "user://test/screenshots/current/{component}_{theme}_{viewport}.png"
    DirAccess.make_dir_absolute("user://test/screenshots/current")
    image.save_png(current_path)

    # Compare with baseline
    var baseline_path = "res://test/screenshots/baseline/{component}_{theme}_{viewport}.png"
    var baseline_image = Image.new()
    if baseline_image.load(baseline_path) == OK:
        var diff = compare_images(image, baseline_image)
        assert_lt(diff, 0.01, "Visual difference should be < 1%")
    else:
        # Create baseline if it doesn't exist
        print("Baseline not found, creating new baseline at: ", baseline_path)
        DirAccess.make_dir_absolute("res://test/screenshots/baseline")
        image.save_png(baseline_path)
```

### Image Comparison Logic

```gdscript
func compare_images(img1: Image, img2: Image) -> float:
    # Calculate pixel difference percentage
    if img1.get_size() != img2.get_size():
        return 1.0  # 100% different if sizes don't match

    var diff_pixels = 0
    var total_pixels = img1.get_width() * img1.get_height()

    for x in range(img1.get_width()):
        for y in range(img1.get_height()):
            var c1 = img1.get_pixel(x, y)
            var c2 = img2.get_pixel(x, y)
            if c1 != c2:
                diff_pixels += 1

    return float(diff_pixels) / float(total_pixels)
```

## Theme Switching Validation

### Theme Consistency Requirements

**Layout invariance:** Theme switching must not change component layout or spacing. Only colors should change.

**Test pattern:**
```gdscript
func test_theme_switching_layout_invariance():
    var theme_manager = ThemeManager.new()
    add_child(theme_manager)

    # Test light theme
    theme_manager.set_theme("light")
    await wait_for_frame()
    var light_screenshot = capture_viewport()

    # Test dark theme
    theme_manager.set_theme("dark")
    await wait_for_frame()
    var dark_screenshot = capture_viewport()

    # Ensure theme switch doesn't break layout
    var layout_diff = compare_layouts(light_screenshot, dark_screenshot)
    assert_eq(layout_diff, 0.0, "Theme switch should not change layout")

    theme_manager.queue_free()
```

### Color Contrast Validation

**Accessibility requirement:** All text must meet WCAG AA contrast ratio (4.5:1 for normal text, 3:1 for large text).

**Test pattern:**
```gdscript
func test_color_contrast_accessibility():
    var text_color = ArcherDesignTokens.get_text_primary_color(is_dark)
    var bg_color = ArcherDesignTokens.get_background_color(is_dark)

    var contrast_ratio = calculate_contrast_ratio(text_color, bg_color)
    assert_gte(contrast_ratio, 4.5, "Text must meet WCAG AA contrast ratio")

func calculate_contrast_ratio(fg: Color, bg: Color) -> float:
    var fg_luminance = calculate_luminance(fg)
    var bg_luminance = calculate_luminance(bg)

    var lighter = max(fg_luminance, bg_luminance)
    var darker = min(fg_luminance, bg_luminance)

    return (lighter + 0.05) / (darker + 0.05)
```

var button = BaseButton.new()
button.add_theme_constant_override("padding_left", ArcherDesignTokens.SPACING_SM)
