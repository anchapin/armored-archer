```bash
#!/bin/bash
set -e

# Gap Analysis Script
# Identifies functions with 0% coverage for prioritized testing

cd "$(dirname "$0")/../.."

# Colors for output
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m' # No Color

echo "=== Coverage Gap Analysis ==="

# Check if coverage file exists
if [ ! -f "coverage/coverage.out" ]; then
    echo -e "${RED}ERROR: coverage/coverage.out not found${NC}"
    echo "Run 'bash scripts/generate-coverage-report.sh' first"
    exit 1
fi

# Generate coverage report
echo ""
echo "Generating coverage report..."
go test -coverprofile=coverage/coverage.out -covermode=atomic ./... > /dev/null 2>&1

# Extract 0% coverage functions
echo ""
echo -e "${YELLOW}Functions with 0% coverage:${NC}"
echo ""

# Count total functions and zero-coverage functions
TOTAL_FUNCS=$(go tool cover -func=coverage/coverage.out | grep -v "^total:" | wc -l)
ZERO_COVERAGE_FUNCS=$(go tool cover -func=coverage/coverage.out | grep -v "^total:" | awk '$3 == "0.0%"' | wc -l)

echo "Total functions: $TOTAL_FUNCS"
echo "Functions with 0% coverage: $ZERO_COVERAGE_FUNCS"
echo ""

# Generate console output grouped by package
go tool cover -func=coverage/coverage.out | grep -v "^total:" | awk '
{
    pkg = $2
    gsub(/\/[^\/]+$/, "", pkg)
    func = $1
    gsub(/^.*:/, "", func)
    coverage = $3

    if (coverage == "0.0%") {
        if (pkg != current_pkg) {
            if (current_pkg != "") {
                print ""
            }
            print "Package: " pkg
            current_pkg = pkg
        }
        print "  " func
    }
}'

# Generate JSON output for programmatic use
echo ""
echo "Generating JSON output..."
OUTPUT_FILE="coverage/gaps.json"
go tool cover -func=coverage/coverage.out | grep -v "^total:" | awk '
BEGIN {
    print "{"
    print "  \"gaps\": ["
    first = 1
    count = 0
}
{
    pkg = $2
    gsub(/\/[^\/]+$/, "", pkg)
    func = $1
    coverage = $3

    if (coverage == "0.0%") {
        if (!first) {
            print ","
        }
        gsub(/^.*\//, "", func)
        printf "    {\"package\": \"%s\", \"function\": \"%s\"}", pkg, func
        first = 0
        count++
    }
}
END {
    print ""
    print "  ],"
    print "  \"total\": " count
    print "}"
}' > "$OUTPUT_FILE"

echo -e "${GREEN}JSON output written to: $OUTPUT_FILE${NC}"

# Summary
echo ""
echo "=== Summary ==="
OVERALL_COVERAGE=$(go tool cover -func=coverage/coverage.out | grep "^total:" | awk '{print $3}')
echo "Overall coverage: $OVERALL_COVERAGE"
echo -e "${RED}Functions needing tests: $ZERO_COVERAGE_FUNCS${NC}"
echo -e "${YELLOW}Priority: Start with functions in critical packages (combat, matchmaking, rpg)${NC}"

exit 0
```
