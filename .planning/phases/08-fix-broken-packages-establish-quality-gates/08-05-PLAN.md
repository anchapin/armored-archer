```bash
#!/bin/bash
set -e

# Coverage Gates Enforcement Script
# Enforces coverage thresholds: 60% overall, 80% critical path

cd "$(dirname "$0")/../.."

# Colors for output
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m' # No Color

OVERALL_THRESHOLD=60.0
CRITICAL_THRESHOLD=80.0
CRITICAL_PACKAGES="combat matchmaking rpg"

echo "=== Coverage Gates Enforcement ==="
echo "Overall threshold: ${OVERALL_THRESHOLD}%"
echo "Critical path threshold: ${CRITICAL_THRESHOLD}%"
echo "Critical packages: ${CRITICAL_PACKAGES}"
echo ""

# Check if coverage file exists
if [ ! -f "coverage/coverage.out" ]; then
    echo -e "${RED}ERROR: coverage/coverage.out not found${NC}"
    exit 1
fi

# Extract overall coverage
OVERALL=$(go tool cover -func=coverage/coverage.out | grep "^total:" | awk '{print $3}' | sed 's/%//')
OVERALL_NUM=$(echo "$OVERALL" | awk '{printf "%.1f", $1}')

echo "Overall coverage: ${OVERALL_NUM}%"

# Check overall threshold
if (( $(echo "$OVERALL_NUM < $OVERALL_THRESHOLD" | bc -l 2>/dev/null || echo "0") )); then
    echo -e "${RED}FAIL: Overall coverage ${OVERALL_NUM}% is below threshold ${OVERALL_THRESHOLD}%${NC}"
    exit 1
fi

echo -e "${GREEN}PASS: Overall coverage ${OVERALL_NUM}% meets threshold${NC}"

# Check critical packages
echo ""
echo "Critical package coverage:"

CRITICAL_MET=1
for pkg in $CRITICAL_PACKAGES; do
    # Extract coverage for specific package
    pkg_lines=$(go tool cover -func=coverage/coverage.out | grep "internal/$pkg/" | grep -v "^total:")

    if [ -z "$pkg_lines" ]; then
        echo -e "  ${YELLOW}internal/$pkg:${NC} No coverage data (not tested yet)"
        continue
    fi

    # Calculate average coverage for this package
    pkg_cov=$(echo "$pkg_lines" | awk '{
        coverage = $3
        gsub(/%/, "", coverage)
        sum += coverage
        count++
    } END {
        if (count > 0) printf "%.1f", sum/count
        else print "0.0"
    }')

    echo -n "  internal/$pkg: ${pkg_cov}%"

    # Check critical threshold
    if (( $(echo "$pkg_cov < $CRITICAL_THRESHOLD" | bc -l 2>/dev/null || echo "1") )); then
        echo -e " ${RED}(FAIL - threshold: ${CRITICAL_THRESHOLD}%)${NC}"
        CRITICAL_MET=0
    else
        echo -e " ${GREEN}(PASS)${NC}"
    fi
done

echo ""

# Exit with failure if any critical package below threshold
if [ $CRITICAL_MET -eq 0 ]; then
    echo -e "${RED}FAIL: One or more critical packages below threshold${NC}"
    exit 1
fi

echo -e "${GREEN}All coverage gates passed${NC}"
exit 0
```
